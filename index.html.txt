  <script>
    // Inicializar Telegram WebApp
    const tg = window.Telegram.WebApp;
    tg.ready();
    tg.expand(); // Expandir a pantalla completa en Telegram

    let wemosIP = localStorage.getItem('wemos_ip') || '192.168.4.1';
    document.getElementById('ip-input').value = wemosIP;

    function guardarIP() {
      wemosIP = document.getElementById('ip-input').value.trim();
      localStorage.setItem('wemos_ip', wemosIP);
      if (tg.HapticFeedback) tg.HapticFeedback.notificationOccurred('success');
      updateData();
    }

    // Inicializar Chart.js
    const ctx = document.getElementById('tempChart').getContext('2d');
    const chart = new Chart(ctx, {
      type: 'line',
      data: {
        labels: [],
        datasets: [
          { label: 'S1 (°C)', borderColor: '#ef4444', data: [], fill: false, tension: 0.2 },
          { label: 'S2 (°C)', borderColor: '#3b82f6', data: [], fill: false, tension: 0.2 }
        ]
      },
      options: {
        responsive: true,
        maintainAspectRatio: false,
        scales: { y: { suggestedMin: -5, suggestedMax: 45 } }
      }
    });

    function updateData() {
      if (!wemosIP) return;
      
      fetch(`http://${wemosIP}/data`)
        .then(res => res.json())
        .then(data => {
          document.getElementById('s1').innerText = data.t1;
          document.getElementById('s2').innerText = data.t2;
          document.getElementById('dt').innerText = data.dt;

          const dtBox = document.getElementById('dt-box');
          if (data.dt >= 20.0) {
            dtBox.className = "dt-box dt-alert";
            document.getElementById('alert-tag').innerText = " (ALTO)";
          } else {
            dtBox.className = "dt-box";
            document.getElementById('alert-tag').innerText = "";
          }

          if (data.hist1 && data.hist2 && data.labels) {
            chart.data.labels = data.labels;
            chart.data.datasets[0].data = data.hist1;
            chart.data.datasets[1].data = data.hist2;
            chart.update();
          }
        })
        .catch(err => console.log("Error conectando al Wemos:", err));
    }

    function descargarCSV() {
      window.location.href = `http://${wemosIP}/download_full`;
    }

    setInterval(updateData, 3000);
    updateData();
  </script>
